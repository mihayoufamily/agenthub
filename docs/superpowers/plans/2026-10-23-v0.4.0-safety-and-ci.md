```typescript
/**
 * 命令串 → argv（MCP server 启动用；不碰 electron，逻辑层可直接测）。
 *
 * 为什么自己切而不是 shell:true：整条命令交 cmd 解析，等于把"配置里写什么就跑什么"变成
 * 任意 shell 语法。切完存成数组，起进程时不再经过 shell 的解释。
 * 为什么留 comspec 一条路：Windows 上 npx/npm 的真身是 .CMD，Node 对 .bat/.cmd
 * 在非 shell 启动下有专门限制，硬撑"纯 shell:false"只会让用户用不了最常见的 MCP server。
 */

/** 一个命令的可执行文件 + 参数 + 是否需要包一层 comspec */
export interface LaunchPlan {
  file: string
  args: string[]
  comspec: boolean
}

/** 引号感知按空白切分：引号内的空格是参数内容，不是分隔符 */
export function tokenizeCommand(command: string): string[] {
  const out: string[] = []
  let cur = ''
  let q: string | null = null
  let started = false
  const push = () => {
    if (started) out.push(cur)
    cur = ''
    started = false
  }
  for (let i = 0; i < (command ?? '').length; i++) {
    const ch = command[i]
    if (q) {
      // 引号内：原样收下，遇到同种引号结束（不认转义，Windows 路径里没有可靠的转义规则）
      if (ch === q) q = null
      else cur += ch
      continue
    }
    if (ch === '"' || ch === "'") {
      q = ch
      started = true
      continue
    }
    if (/\s/.test(ch)) {
      push()
      continue
    }
    cur += ch
    started = true
  }
  // 引号不成对：把已读内容原样作为一段交出去（用户能立刻在预览里看到切错了）
  push()
  return out
}

/** .cmd / .bat 家族，以及 npm 系裸启动器：这些必须经 comspec 才起得来 */
export function needsComspec(bin: string): boolean {
  const b = (bin ?? '').trim()
  if (/\.(cmd|bat)$/i.test(b)) return true
  // npx/npm/pnpm/yarn 的真身是 .CMD，PATH 上没有同名 .exe；shell:false 直接 spawn 会 ENOENT。
  // node / node.exe / 带路径的可执行文件仍走 argv 直传。
  return !/[/\\]/.test(b) && /^(npx|npm|pnpm|yarn)$/i.test(b)
}

/** 由命令串算出启动计划；切不出可执行文件时返回 null，由调用方如实报错（不起空进程） */
export function toLaunchPlan(command: string): LaunchPlan | null {
  const parts = tokenizeCommand(command)
  if (!parts.length) return null
  const [file, ...args] = parts
  if (!file.trim()) return null // 空引号（"" / " "）切得出空 token，不能当可执行文件交出去
  return { file, args, comspec: needsComspec(file) }
}
```

```typescript
// 配置迁移的纯规则层（store.ts 只负责读写盘；规则放这里才能被逻辑层验证覆盖）
import type { McpServer } from './types'

/** 当前配置格式头版本：`store.ts` 的迁移链与 importConfig 都以此为准，不再各写各的数字 */
export const SCHEMA_HEAD = 2

/** v1 -> v2 只关心这三项；其余字段原样透传，不做全量 AppConfig 断言 */
export interface McpConfigShape {
  mcpServers?: McpServer[]
  sandboxFallback?: 'auto' | 'ask' | 'off'
  schemaVersion?: number
}

/**
 * MCP 启动方式从"整条 shell 命令"改成 argv。
 * 只有 command 没有 launch 的记录一律标失效、禁用、保留原文——不猜着拆（拆错的命令行比"连不上"更难定位），
 * 也不删（用户重填时至少知道原来填的是什么）。这是用户 2026-10-23 拍定的。
 */
export function migrateV1toV2(raw: McpConfigShape): McpConfigShape {
  const out: McpConfigShape = { ...raw }
  if (Array.isArray(raw.mcpServers)) {
    out.mcpServers = raw.mcpServers.map((s) => {
      // 脏数据（null / 字符串 / 数字）原样透传：{ ...s } 会把它们伪装成记录并写回 config.json
      if (!s || typeof s !== 'object') return s
      const bin = typeof s.launch?.bin === 'string' ? s.launch.bin.trim() : ''
      return bin ? { ...s, legacyInvalid: false, formatVersion: SCHEMA_HEAD as 2 } : { ...s, legacyInvalid: true, enabled: false }
    })
  }
  if (out.sandboxFallback !== 'auto' && out.sandboxFallback !== 'ask' && out.sandboxFallback !== 'off') {
    out.sandboxFallback = 'auto'
  }
  return out
}
```

/** 旧版单命令配置的拒收码：界面按这个码渲染，不靠文案匹配 */
export const MCP_LEGACY_CODE = 'invalid_legacy_command'

/**
 * 把配置换算成一次 spawn 的入参（纯裁决，逻辑层可直接断言）。
 * 三种结果：
 * ① 有 launch → 数组直传，shell:false；
 * ② 只有旧 command → 抛 legacy 错误码，等用户重填（不猜拆，也不删记录）；
 * ③ npm 家族与 .cmd/.bat（由 needsComspec 判定）→ 显式经 cmd.exe，参数走拒绝式构造。②是用户 2026-10-23 拍定的，③是 Windows 的现实。
 */
export interface SpawnPlan {
  file: string
  args: string[]
  options: { windowsHide: boolean; shell?: boolean; windowsVerbatimArguments?: boolean }
  /** 这条真的经 cmd.exe 解释——界面与文档要按它标注，不许用 shell 布尔代替（评审后改） */
  comspec: boolean
}

/** cmd 的元字符：不在引号内就会被当成命令分隔或重定向 */
const CMD_META = /[&|^<>]/

/**
 * comspec 分支的参数构造：只允许"能安全表达"的 token。
 * cmd 里 `\"` 不是转义（反斜杠原样保留、引号照样闭合），所以含 `"` 的 token 根本无法安全表示；
 * `%` 在引号内仍会展开。两者一律拒绝，而不是假装转义——评审用真实 repro 证明旧写法能把
 * `a"b & type nul > f` 变成第二条命令。含空格的 token 加引号后 cmd 会把 `& | ^ < >` 当字面量。
 */
function forCmd(token: string): string {
  if (/["%]/.test(token)) throw new Error(`MCP 参数含 cmd 无法安全表示的字符（" 或 %）：${token.slice(0, 40)}`)
  if (!/\s/.test(token) && CMD_META.test(token)) throw new Error(`MCP 参数含 cmd 元字符且无法加引号保护：${token.slice(0, 40)}`)
  return /\s/.test(token) ? `"${token}"` : token
}

export function planSpawn(server: McpServer): SpawnPlan {
  const bin = (server.launch?.bin ?? '').trim()
  const rawArgs = Array.isArray(server.launch?.args) ? server.launch.args : []
  if (rawArgs.some((a) => typeof a !== 'string')) throw new Error('MCP server 的参数里有非字符串项，已拒绝（不静默丢弃）')
  const args = rawArgs as string[]
  if (!bin) {
    // 没 launch 只有 command：旧格式。v0.4.0 明确不作废式兼容——不自动拆分
    if ((server.command ?? '').trim()) throw new Error(`${MCP_LEGACY_CODE}：该 server 是旧版单命令配置，已停用，请在设置里重填`)
    throw new Error('MCP server 没有启动命令')
  }
  if (needsComspec(bin)) {
    const line = [bin, ...args].map(forCmd).join(' ')
    const comspecExe = process.env.ComSpec || 'C:\\Windows\\System32\\cmd.exe'
    // windowsVerbatimArguments：Node 不再改写引号。实测 {shell:false} 直调 cmd.exe 可行且无 DEP0190，
    // 而 shell:true 会让 Node 再包一层（/d /s /c "cmd.exe /d /c ..."）并打出"参数未转义"告警。
    return { file: comspecExe, args: ['/d', '/c', line], options: { windowsHide: true, windowsVerbatimArguments: true }, comspec: true }
  }
  return { file: bin, args, options: { windowsHide: true }, comspec: false }
}

```typescript
/**
 * 窗口导航裁决（纯函数，逻辑层可测）。主进程三个 handler（新窗口 / 导航 / 重定向）共用它，
 * 免得规则在 Electron 回调里写三份、改一漏二。
 * 现状：渲染层没有任何 target=_blank 或 window.open（全量 grep 0 命中），这里是防退化兜底。
 */
export type NavAction = 'allow' | 'external' | 'block'

/** 本应用自己的页面：生产是 resources/app/dist/renderer 下的 file://，开发是 VITE_DEV_SERVER_URL */
function isOwnPage(url: string, devServerUrl: string | null): boolean {
  if (devServerUrl) {
    try {
      const target = new URL(url)
      const dev = new URL(devServerUrl)
      if (target.protocol === dev.protocol && target.host === dev.host) return true
    } catch {
      return false
    }
  }
  // file:// 只认渲染产物目录，C:\Windows\... 这类本地文件不该在窗口里加载
  return /^file:\/\/\/[^?#]*\/resources\/app\/dist\/renderer\//i.test(url)
}

export function navDecision(url: string, devServerUrl: string | null): NavAction {
  const u = (url ?? '').trim()
  if (!u) return 'block'
  let parsed: URL
  try {
    parsed = new URL(u)
  } catch {
    return 'block'
  }
  const p = parsed.protocol
  if (p === 'file:') return isOwnPage(u, devServerUrl) ? 'allow' : 'block'
  if (p === 'http:' || p === 'https:') return isOwnPage(u, devServerUrl) ? 'allow' : 'external'
  // 其余协议（javascript:、data:、file 之外的自定义 scheme）一律不进窗口、也不丢给系统
  return 'block'
}
```

/**
 * 遮掉看起来像凭据的片段。Hook 命令常有人直接写 `--api-key=...`，
 * 界面回显与账本都会带上它，指纹显示前必须先脱敏。
 */
export function redactSecrets(s: string): string {
  return (s ?? '')
    .replace(/((?:api[_-]?key|token|secret|password|passwd|pwd)[=\s]+)([^\s"']{4,})/gi, '$1***')
    .replace(/(Bearer\s+)([A-Za-z0-9._\-]{6,})/gi, '$1***')
}

/** 12 位指纹：只给人肉眼比对"这条命令变没变过"，不参与任何放行判定 */
export function fingerprintCommand(s: string): string {
  const base = normalizeCommand(redactSecrets(s))
  // 与项目里其它 id 生成同思路：不用 crypto（shared 层要在纯 node 与浏览器两端同构可用）
  let h1 = 0x811c9dc5
  let h2 = 0x1000193
  for (let i = 0; i < base.length; i++) {
    h1 = (h1 ^ base.charCodeAt(i)) >>> 0
    h1 = Math.imul(h1, 0x01000193) >>> 0
    h2 = (Math.imul(h2 ^ base.charCodeAt(i), 0x85ebca6b)) >>> 0
  }
  return (h1.toString(36) + h2.toString(36)).slice(0, 12).padEnd(12, '0')
}
